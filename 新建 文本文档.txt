<p class="font-body-reading text-body-reading text-reading-ink mb-6">
                            Hi HN, I wanted to share a weekend project I've been working on. Like many here, I've struggled to find an RSS reader that strikes the right balance between simplicity and functionality since the demise of Google Reader.
                        </p>
<div class="w-full h-64 bg-surface-container rounded-lg mb-6 overflow-hidden fluent-elevation relative">

</div>
<h2 class="font-title-md text-title-md text-on-surface mt-8 mb-4">Why another reader?</h2>
<p class="font-body-reading text-body-reading text-reading-ink mb-6">
                            Most modern readers feel either too bloated with social features or too spartan. I wanted something that felt native, respected my system theme, and got out of the way of the content.
                        </p>
<h3 class="font-subtitle text-subtitle text-on-surface font-semibold mt-6 mb-3">核心功能：</h3>
<ul class="list-disc pl-6 mb-6 font-body-reading text-body-reading text-reading-ink space-y-2">
<li class="">Local-first architecture for speed</li>
<li class="">OPML import/export</li>
<li class="">Keyboard shortcuts for navigation (j/k)</li>
<li class="">Reader view for truncated feeds</li>
</ul>
<p class="font-body-reading text-body-reading text-reading-ink mb-6">
                            The stack is just vanilla JS, Tailwind CSS for styling, and a tiny local database. No complex build steps or heavy frameworks.
                        </p>
<div class="bg-surface-container-low p-4 rounded-md border-l-4 border-primary mb-6">

                                };
                            </p>
</div>
<p class="font-body-reading text-body-reading text-reading-ink">

        // Reader UI Toggle Logic
        const body = document.body;
        const toggleZone = document.getElementById('ui-toggle-zone');
        let uiVisible = true;

        toggleZone.addEventListener('click', () => {
            uiVisible = !uiVisible;
            if (uiVisible) {
                body.classList.remove('reader-ui-hidden');
                document.getElementById('reader-bottom-bar').classList.remove('translate-y-full');
            } else {
                body.classList.add('reader-ui-hidden');
                document.getElementById('reader-bottom-bar').classList.add('translate-y-full');
                hideSettings();
            }
        });

        // Settings Panel Logic
        const btnSettings = document.getElementById('btn-settings');
        const settingsPanel = document.getElementById('settings-panel');
        let settingsVisible = false;

        btnSettings.addEventListener('click', (e) => {
            e.stopPropagation();
            settingsVisible = !settingsVisible;
            if (settingsVisible) {
                settingsPanel.classList.remove('hidden');
                // Small delay for transition to work after display:block
                setTimeout(() => {
                    settingsPanel.classList.remove('opacity-0', 'translate-y-[-10px]');
                    settingsPanel.classList.add('opacity-100', 'translate-y-0');
                }, 10);
            } else {
                hideSettings();
            }
        });

        function hideSettings() {
            settingsVisible = false;
            settingsPanel.classList.remove('opacity-100', 'translate-y-0');
            settingsPanel.classList.add('opacity-0', 'translate-y-[-10px]');
            setTimeout(() => {
                settingsPanel.classList.add('hidden');
            }, 200);
        }

        // Close settings when clicking outside
        document.addEventListener('click', (e) => {
            if (settingsVisible && !settingsPanel.contains(e.target) && e.target !== btnSettings) {
                hideSettings();
            }
        });

        // Theme switching logic
        const readingContainer = document.getElementById('reading-container');
        const documentContent = document.getElementById('document-content');
        
        function setTheme(theme) {
            // Reset classes
            readingContainer.className = 'h-screen w-full flex items-center justify-center relative transition-colors duration-500';
            
            // Remove active state from all theme buttons
            const buttons = settingsPanel.querySelectorAll('button');
            buttons.forEach(btn => {
                btn.classList.remove('border-primary', 'ring-2', 'ring-primary/20');
                btn.classList.add('border-outline-variant/50');
            });
            
            // Apply new theme
            if (theme === 'light') {
                readingContainer.classList.add('bg-surface');
                documentContent.querySelectorAll('p').forEach(p => p.className = 'mb-6 indent-8 text-reading-ink');
                documentContent.querySelector('h1').className = 'font-display text-display mb-8 mt-4 text-on-surface';
                event.target.classList.add('border-primary', 'ring-2', 'ring-primary/20');
                event.target.classList.remove('border-outline-variant/50');
            } else if (theme === 'sepia') {
                readingContainer.classList.add('bg-reading-paper');
                documentContent.querySelectorAll('p').forEach(p => p.className = 'mb-6 indent-8 text-[#433422]');
                documentContent.querySelector('h1').className = 'font-display text-display mb-8 mt-4 text-[#2c2216]';
                event.target.classList.add('border-primary', 'ring-2', 'ring-primary/20');
                event.target.classList.remove('border-outline-variant/50');
            } else if (theme === 'dark') {
                readingContainer.classList.add('bg-inverse-surface');
                documentContent.querySelectorAll('p').forEach(p => p.className = 'mb-6 indent-8 text-inverse-on-surface opacity-90');
                documentContent.querySelector('h1').className = 'font-display text-display mb-8 mt-4 text-inverse-on-surface');
                event.target.classList.add('border-primary', 'ring-2', 'ring-primary/20');
                event.target.classList.remove('border-outline-variant/50');
            }
        }

        // Font size logic
        const fontSizeSlider = document.getElementById('font-size-slider');
        
        function updateFontSize(size) {
            documentContent.style.fontSize = `${size}px`;
            // Adjust line height proportionally for better readability
            documentContent.style.lineHeight = `${size * 1.6}px`;
        }

        fontSizeSlider.addEventListener('input', (e) => {
            updateFontSize(e.target.value);
        });

        function changeFontSize(delta) {
            let currentSize = parseInt(fontSizeSlider.value);
            let newSize = currentSize + (delta * 2);
            if (newSize >= 14 && newSize <= 32) {
                fontSizeSlider.value = newSize;
                updateFontSize(newSize);
            }
        }

        // Dummy page turn function
        function turnPage(direction) {
            const article = document.getElementById('document-content');
            const scrollAmount = article.clientHeight * 0.9;
            
            if (direction === 'next') {
                article.scrollBy({ top: scrollAmount, behavior: 'smooth' });
            } else {
                article.scrollBy({ top: -scrollAmount, behavior: 'smooth' });
            }
            
            // Auto hide UI on page turn if visible
            if (uiVisible) {
                body.classList.add('reader-ui-hidden');
                document.getElementById('reader-bottom-bar').classList.add('translate-y-full');
                hideSettings();
                uiVisible = false;
            }
        }

<script>
        let isChromeVisible = true;
        const topBar = document.getElementById('top-bar');
        const bottomBar = document.getElementById('bottom-bar');

        function toggleChrome() {
            isChromeVisible = !isChromeVisible;
            if (isChromeVisible) {
                topBar.classList.remove('-translate-y-full');
                bottomBar.classList.remove('translate-y-full');
            } else {
                topBar.classList.add('-translate-y-full');
                bottomBar.classList.add('translate-y-full');
            }
        }
        
        // Auto-hide after 3 seconds on load
        setTimeout(() => {
            if(isChromeVisible) toggleChrome();
        }, 3000);
    </script>
